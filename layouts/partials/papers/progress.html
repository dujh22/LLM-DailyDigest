{{- /* 复现工作进度（content/papers/<研究>/<类别>/<短语>/index.md 的 leaf bundle）。 里程碑按顺序累计，前一档未达成则后续不计： 0% 已注册 20% paper/ 下中英文论文齐全（文件名以 en / zh 结尾，如 en.pdf、zh.pdf；或 front matter paper_en / paper_zh 外链） 40% resources/ 已整理（数据 / 代码 / 模型索引，非占位文件） 60% repro/ 有个人复现新增内容 80% front matter reused_by 记录了被哪些 content/research 研究二次利用 100% 已达 80% 且笔记超过 365 天无变化（以 index.md 的 git 最后修改时间为准） 返回 dict：percent / steps（每档 name、done）/ next（下一步说明，100% 时为空） */ -}} {{- $p := . -}} {{- $real := slice -}} {{- range $p.Resources -}} {{- $base := path.Base .Name -}} {{- if not (or (hasPrefix $base ".") (eq $base "index.md")) -}}{{- $real = $real | append .Name -}}{{- end -}} {{- end -}} {{- $en := or $p.Params.paper_en (gt (len ($p.Resources.Match "paper/*en.*")) 0) -}} {{- $zh := or $p.Params.paper_zh (gt (len ($p.Resources.Match "paper/*zh.*")) 0) -}} {{- $res := false -}}{{- $rep := false -}} {{- range $real -}} {{- if hasPrefix .
